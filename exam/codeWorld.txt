module Task21 (main, balloon) where

import CodeWorld
import Prelude hiding (($))

balloon :: Double -> Picture
balloon t
    | t < 5 =
        colored green (solidCircle (0.5 * t))

    | t < 8 =
        colored yellow (solidCircle (2.5 - 0.5 * (t - 5)))

    | otherwise =
        colored red (solidCircle 1)

main :: IO ()
main = animationOf balloon
